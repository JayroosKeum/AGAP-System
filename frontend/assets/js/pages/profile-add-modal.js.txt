document.addEventListener('DOMContentLoaded', () => {
    const modal = document.getElementById('addProfileModal'); if (!modal) return;
    const form = document.getElementById('addProfileForm'), address = document.getElementById('profileAddress'), purok = document.getElementById('profilePurok'), lat = document.getElementById('profileLatitude'), lng = document.getElementById('profileLongitude'), wrap = document.getElementById('profileMapWrap'), toggle = document.getElementById('toggleProfileMap'), alertBox = document.getElementById('addProfileAlert'), save = document.getElementById('saveProfileButton');
    let map, marker, boundary, lastLookup = 0, addressTimer; const center = [14.6548, 121.0862];
    const purokCenters = { 'Tumana': [14.6548, 121.0862], 'Doña Petra': [14.6558, 121.0824], 'Purok 1': [14.6623, 121.0804], 'Purok 2': [14.6602, 121.0830], 'Purok 3': [14.6585, 121.0857], 'Purok 4': [14.6564, 121.0877], 'Purok 5': [14.6538, 121.0888], 'Purok 6': [14.6514, 121.0867], 'Purok 7': [14.6493, 121.0834] };
    function open() { modal.hidden = false; modal.setAttribute('aria-hidden', 'false'); document.body.classList.add('profile-modal-open'); setTimeout(initMap, 0); form.elements.first_name.focus() }
    function close() { modal.hidden = true; modal.setAttribute('aria-hidden', 'true'); document.body.classList.remove('profile-modal-open') }
    document.querySelectorAll('[data-open-profile-modal]').forEach(b => b.onclick = open); modal.querySelectorAll('[data-close-profile-modal]').forEach(b => b.onclick = close); document.addEventListener('keydown', e => { if (e.key === 'Escape' && !modal.hidden) close() });
    function setError(input, message) { input.classList.toggle('is-invalid', !!message); const e = form.querySelector(`[data-error-for="${input.name}"]`); if (e) e.textContent = message || '' }
    function validate(input) { if (input.required && !String(input.value).trim()) { setError(input, 'This field is required.'); return false } if (input.type === 'email' && input.value && !input.validity.valid) { setError(input, 'Enter a valid email address.'); return false } setError(input, ''); return true }
    form.querySelectorAll('input:not([type="hidden"]),select,textarea').forEach(input => { input.addEventListener('blur', () => validate(input)); input.addEventListener('change', () => validate(input)) });
    function initMap() {
        if (map) { map.invalidateSize(); return } if (typeof L === 'undefined') { alertBox.hidden = false; alertBox.textContent = 'The map library could not load. Reload the page before saving.'; return }
        map = L.map('profileMap', { minZoom: 13, maxZoom: 19 }).setView(center, 15); L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', { maxZoom: 19, attribution: '© OpenStreetMap contributors' }).addTo(map);
        const shape = [[14.6713, 121.0780], [14.6680, 121.0868], [14.6637, 121.0917], [14.6568, 121.0940], [14.6490, 121.0917], [14.6440, 121.0849], [14.6462, 121.0767], [14.6530, 121.0731], [14.6610, 121.0732]]; boundary = L.polygon(shape, { color: '#2563eb', weight: 3, fillColor: '#60a5fa', fillOpacity: .18 }).addTo(map); map.fitBounds(boundary.getBounds(), { padding: [18, 18] }); marker = L.marker(center, { draggable: true }).addTo(map); marker.on('dragend', () => reversePin(marker.getLatLng())); map.on('click', e => { marker.setLatLng(e.latlng); reversePin(e.latlng) });
    }
    function positionPin(position, zoom = 17) { if (!map) initMap(); if (!map || !marker) return; marker.setLatLng(position); map.setView(position, zoom); lat.value = Number(position[0]).toFixed(7); lng.value = Number(position[1]).toFixed(7) }
    async function reversePin(pos) { positionPin([pos.lat, pos.lng]); const label = document.getElementById('pinInstruction'); label.textContent = 'Finding address...'; try { const r = await fetch(`backend/api/geocoding/reverse.php?lat=${encodeURIComponent(pos.lat)}&lng=${encodeURIComponent(pos.lng)}`, { cache: 'no-store', headers: { Accept: 'application/json' } }), d = await r.json(); if (!d.success) throw new Error(d.message || 'Address lookup failed.'); if (d.address) address.value = d.address; if (d.purok) { let option = [...purok.options].find(o => o.value.toLowerCase() === String(d.purok).toLowerCase()); if (!option) { option = new Option(d.purok, d.purok); purok.add(option) } purok.value = option.value } else if (!purok.value) { purok.value = 'Tumana' } validate(address); validate(purok); alertBox.hidden = true } catch (e) { alertBox.hidden = false; alertBox.textContent = 'The pin was updated. Complete the address and Purok manually.' } finally { label.textContent = 'Click map to pin' } }
    async function forwardAddress() { if (!validate(address)) return; const label = document.getElementById('pinInstruction'); label.textContent = 'Locating address...'; try { const r = await fetch(`backend/api/geocoding/forward.php?q=${encodeURIComponent(address.value)}`, { cache: 'no-store', headers: { Accept: 'application/json' } }), d = await r.json(); if (!d.success) throw new Error(d.message || 'Address could not be located.'); positionPin([d.latitude, d.longitude]); alertBox.hidden = true } catch (e) { alertBox.hidden = false; alertBox.textContent = e.message } finally { label.textContent = 'Click map to pin' } }
    purok.addEventListener('change', () => { const position = purokCenters[purok.value]; if (position) { positionPin(position, 16); const typed = address.value.trim(); if (!typed) address.value = `${purok.value}, Barangay Tumana, Marikina City`; validate(address) } });
    address.addEventListener('change', forwardAddress); address.addEventListener('blur', () => { clearTimeout(addressTimer); addressTimer = setTimeout(forwardAddress, 300) });
    toggle.onclick = () => { const hidden = wrap.hidden = !wrap.hidden; toggle.textContent = hidden ? '⌾ Show Map' : '⌾ Hide Map'; if (!hidden) setTimeout(() => map && map.invalidateSize(), 0) };
    form.onsubmit = async event => { event.preventDefault(); const visible = [...form.querySelectorAll('input:not([type="hidden"]),select,textarea')]; const valid = visible.map(validate).every(Boolean); if (!valid) { visible.find(i => i.classList.contains('is-invalid'))?.focus(); return } if (!lat.value || !lng.value) { alertBox.hidden = false; alertBox.textContent = 'Position the map pin by selecting a Purok, entering an address, or clicking the map.'; return } save.disabled = true; save.textContent = 'Saving...'; alertBox.hidden = true; try { const r = await fetch(form.action, { method: 'POST', body: new FormData(form), headers: { Accept: 'application/json' } }), d = await r.json(); if (!d.success) { for (const [name, message] of Object.entries(d.errors || {})) { const input = form.elements[name]; if (input) setError(input, message) } throw new Error(d.message || 'Unable to save profile.') } close(); location.href = `index.php?page=profiles&action=view&id=${d.id}` } catch (error) { alertBox.hidden = false; alertBox.textContent = error.message } finally { save.disabled = false; save.textContent = 'Save Profile' } };
});
