document.addEventListener('DOMContentLoaded', () => {
    const root = document.querySelector('[data-profiles-app]'); if (!root) return; const tbody = root.querySelector('tbody'), summary = root.querySelector('[data-summary]'), pager = root.querySelector('[data-pager]'); let rows = [], page = 1; const perPage = 25;
    const esc = v => { const e = document.createElement('span'); e.textContent = v ?? ''; return e.innerHTML }; const age = d => { if (!d) return 'Age not recorded'; const b = new Date(d + 'T00:00:00'), n = new Date(); let a = n.getFullYear() - b.getFullYear(); if (n < new Date(n.getFullYear(), b.getMonth(), b.getDate())) a--; return a + ' years old' };
    function initials(r) { return ((r.first_name || '').charAt(0) + (r.last_name || '').charAt(0)).toUpperCase() }
    function render() { const total = rows.length, pages = Math.max(1, Math.ceil(total / perPage)); page = Math.min(page, pages); const start = (page - 1) * perPage, part = rows.slice(start, start + perPage); tbody.innerHTML = part.length ? part.map(r => `<tr><td><span class="profile-chip">#RP-${String(r.profile_id).padStart(4, '0')}</span></td><td><div class="person-cell"><span class="avatar">${esc(initials(r))}</span><div><strong>${esc([r.first_name, r.middle_name, r.last_name, r.suffix].filter(Boolean).join(' '))}</strong><small>${esc(r.sex)} · ${esc(r.civil_status)} · ${esc(age(r.birth_date))}</small></div></div></td><td><span class="purok-pill">${esc(r.purok_sitio || 'No purok')}</span><small>${esc(r.address_line)}</small></td><td>${esc(r.contact_no || 'Not provided')}<small>${r.email ? `<a href="mailto:${esc(r.email)}">${esc(r.email)}</a>` : 'No email'}</small></td><td><span class="residency-pill">${esc(r.residency_status)}</span></td><td><a href="index.php?page=profiles&action=view&id=${r.profile_id}">View</a> <a href="index.php?page=profiles&action=form&id=${r.profile_id}">Edit</a></td></tr>`).join('') : '<tr><td colspan="6" class="empty-cell">No profiles found.</td></tr>'; summary.textContent = total ? `Showing ${start + 1}-${Math.min(start + perPage, total)} of ${total} profiles` : 'No profiles'; pager.innerHTML = '';[['First', 1], ['Prev', Math.max(1, page - 1)]].concat(Array.from({ length: pages }, (_, i) => [String(i + 1), i + 1]), [['Next', Math.min(pages, page + 1)], ['Last', pages]]).forEach(([label, target]) => { if (pages > 12 && /^\d+$/.test(label) && Math.abs(Number(label) - page) > 2 && ![1, pages].includes(Number(label))) return; const b = document.createElement('button'); b.type = 'button'; b.textContent = label; b.className = target === page ? 'active' : ''; b.onclick = () => { page = target; render(); root.querySelector('table').scrollIntoView({ behavior: 'smooth', block: 'start' }) }; pager.appendChild(b) }) }
    async function load() { const params = new URLSearchParams(new FormData(root.querySelector('[data-filter-form]'))); const res = await fetch('backend/api/profiles/list.php?' + params, { cache: 'no-store' }), data = await res.json(); rows = data.rows || []; page = 1; for (const [k, v] of Object.entries(data.metrics || {})) { const e = root.querySelector(`[data-metric="${k}"]`); if (e) e.textContent = v ?? 0 } const sel = root.querySelector('[name="purok"]'); if (sel && sel.options.length === 1) (data.puroks || []).forEach(v => sel.add(new Option(v, v))); render() }
    let timer; root.querySelectorAll('input,select').forEach(e => e.addEventListener(e.tagName === 'INPUT' ? 'input' : 'change', () => { clearTimeout(timer); timer = setTimeout(load, 250) })); root.querySelector('[data-reset]').onclick = () => { root.querySelector('[data-filter-form]').reset(); load() }; root.querySelectorAll('[data-quick]').forEach(e => e.onclick = () => { root.querySelector('[name="residency"]').value = e.dataset.quick; load() }); load()
});
